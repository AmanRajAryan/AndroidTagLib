private void showMetadataDialog(String filePath) {
    if (filePath == null || filePath.isEmpty()) {
        Toast.makeText(this, "File path is empty", Toast.LENGTH_SHORT).show();
        return;
    }

    TagLibJNI tagLib = new TagLibJNI();
    HashMap<String, String> metadataMap = null;

    try {
        metadataMap = tagLib.getAllMetadata(filePath);
    } catch (Exception e) {
        Toast.makeText(this, "Failed to read metadata: " + e.getMessage(), Toast.LENGTH_SHORT)
            .show();
        e.printStackTrace();
        return;
    }

    if (metadataMap == null || metadataMap.isEmpty()) {
        Toast.makeText(this, "No metadata found", Toast.LENGTH_SHORT).show();
        return;
    }

    // Build metadata string dynamically
    StringBuilder sb = new StringBuilder();
    for (String key : metadataMap.keySet()) {
        sb.append(key).append(": ").append(metadataMap.get(key)).append("\n");
    }

    // Show AlertDialog
    new AlertDialog.Builder(this)
        .setTitle("Metadata")
        .setMessage(sb.toString())
        .setPositiveButton("OK", null)
        .show();
}
